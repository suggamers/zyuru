// === 音声読み上げエンジンの安定化 === let currentAudio = null; const ZUNDAMON_STYLES = [3, 1, 7, 5, 22, 38, 75, 76]; let currentSpeakId = 0; let currentAbortController = null; // APIリクエストを即座にキャンセルするためのコントローラー function forceResetUI(btnElement, spanEl) { if (btnElement) { btnElement.classList.remove('loading', 'speaking'); if (spanEl) { spanEl.textContent = '聴く'; } } } function stopSpeak() { currentSpeakId++; // IDを更新して進行中の非同期処理を無効化 // 進行中のAPIリクエストがあれば即座にキャンセル if (currentAbortController) { currentAbortController.abort(); currentAbortController = null; } // 再生中の音声があれば即座に停止して破棄 if (currentAudio) { currentAudio.pause(); currentAudio.removeAttribute('src'); currentAudio = null; } // UIを初期状態に戻す document.querySelectorAll('.btn-speak').forEach(btn => { btn.classList.remove('speaking', 'loading'); const span = btn.querySelector('span'); if (span) span.textContent = '聴く'; }); } async function speakText(text, btnElement) { if (!text) return; // 新しい音声を再生する前に、完全にリセットする stopSpeak(); currentSpeakId++; const mySpeakId = currentSpeakId; currentAbortController = new AbortController(); // 新しいキャンセルコントローラーを作成 let spanEl = null; if (btnElement) { btnElement.classList.add('loading'); spanEl = btnElement.querySelector('span'); if (spanEl) spanEl.textContent = '生成中...'; } const currentSpeakerId = ZUNDAMON_STYLES[Math.floor(Math.random() * ZUNDAMON_STYLES.length)]; try { const url = `https://api.tts.quest/v3/voicevox/synthesis?text=${encodeURIComponent(text)}&speaker=${currentSpeakerId}`; // signalを渡すことで、途中でabort()されたらfetchを中断させる const res = await fetch(url, { signal: currentAbortController.signal }); if (mySpeakId !== currentSpeakId) return; // fetch中に別の再生が呼ばれたら中断 const data = await res.json(); if (!data.success) throw new Error("APIエラー"); if (mySpeakId !== currentSpeakId) return; currentAudio = new Audio(); currentAudio.preservesPitch = false; if ('webkitPreservesPitch' in currentAudio) currentAudio.webkitPreservesPitch = false; // ピッチと速度のランダム化は維持 currentAudio.playbackRate = 0.8 + Math.random() * 0.4; currentAudio.onplay = () => { if (mySpeakId !== currentSpeakId) return; if (btnElement) { btnElement.classList.remove('loading'); btnElement.classList.add('speaking'); if (spanEl) spanEl.textContent = '再生中'; } }; currentAudio.onended = () => { if (mySpeakId !== currentSpeakId) return; if (btnElement) forceResetUI(btnElement, spanEl); currentAudio = null; }; currentAudio.onerror = () => { if (mySpeakId !== currentSpeakId) return; if (btnElement) forceResetUI(btnElement, spanEl); }; currentAudio.src = data.mp3DownloadUrl + "?t=" + new Date().getTime(); currentAudio.load(); await currentAudio.play(); } catch (err) { // AbortError（ユーザーが新しくボタンを押してキャンセルされた場合）は無視する if (err.name === 'AbortError') return; if (mySpeakId !== currentSpeakId) return; console.warn("API failed.", err); // APIエラー時もUIを元に戻すだけ（代替音声は流さない） forceResetUI(btnElement, spanEl); } }
